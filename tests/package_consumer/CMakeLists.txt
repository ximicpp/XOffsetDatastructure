cmake_minimum_required(VERSION 3.13)
project(XOffsetExternalConsumer LANGUAGES CXX)
enable_testing()
# Deliberately lower than the full target's requirement; its interface must win.
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(XOFFSET_CONSUMER_COMPONENT storage CACHE STRING "storage or full")
if(DEFINED XOFFSET_SOURCE)
    add_subdirectory("${XOFFSET_SOURCE}" dependency)
    foreach(option XOFFSET_BUILD_TESTS XOFFSET_BUILD_TOOLS XOFFSET_BUILD_EXAMPLES XOFFSET_INSTALL)
        if(${option})
            message(FATAL_ERROR "Subproject unexpectedly enabled ${option}")
        endif()
    endforeach()
    foreach(target test_vm_pool export_signatures helloworld)
        if(TARGET ${target})
            message(FATAL_ERROR "Subproject unexpectedly created ${target}")
        endif()
    endforeach()
else()
    find_package(XOffsetDatastructure CONFIG REQUIRED COMPONENTS ${XOFFSET_CONSUMER_COMPONENT})
endif()
if(XOFFSET_CONSUMER_COMPONENT STREQUAL "full")
    set(interface_target XOffset::xoffset)
else()
    set(interface_target XOffset::storage)
endif()
add_executable(package_consumer main.cpp create.cpp mutate.cpp)
target_link_libraries(package_consumer PRIVATE ${interface_target})
if(XOFFSET_CONSUMER_COMPONENT STREQUAL "full")
    target_compile_definitions(package_consumer PRIVATE XOFFSET_CONSUMER_FULL=1)
endif()
if(MSVC OR CMAKE_CXX_SIMULATE_ID STREQUAL "MSVC")
    target_compile_options(package_consumer PRIVATE /UNDEBUG)
else()
    target_compile_options(package_consumer PRIVATE -UNDEBUG)
endif()
add_test(NAME consumer_transfer COMMAND package_consumer transfer)
add_test(NAME consumer_pool_handle_before_owner COMMAND package_consumer lifetime)
# Owner destruction occurs after main, in another translation unit.
add_test(NAME consumer_static_destruction COMMAND package_consumer static)
