cmake_minimum_required(VERSION 3.30)
project(MemmoveSerialization LANGUAGES CXX)
set(XOFFSET_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(XOFFSET_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(XOFFSET_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(XOFFSET_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../.." "${CMAKE_CURRENT_BINARY_DIR}/library")
add_executable(bench_memmove bench.cpp)
target_link_libraries(bench_memmove PRIVATE XOffset::xoffset)
set_target_properties(bench_memmove PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin/$<CONFIG>")
add_executable(bench_borrowed bench_borrowed.cpp)
target_link_libraries(bench_borrowed PRIVATE XOffset::xoffset)
set_target_properties(bench_borrowed PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin/$<CONFIG>")
add_executable(slg_image_smoke slg_image_smoke.cpp)
target_link_libraries(slg_image_smoke PRIVATE XOffset::xoffset)
target_compile_options(slg_image_smoke PRIVATE -UNDEBUG)
target_include_directories(slg_image_smoke PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../bench/slg/include")
set_target_properties(slg_image_smoke PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin/$<CONFIG>")
