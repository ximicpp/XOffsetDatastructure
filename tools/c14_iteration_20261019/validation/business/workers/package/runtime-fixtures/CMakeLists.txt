cmake_minimum_required(VERSION 3.30)
project(FourPathOptionalRuntimeConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(DEFINED XC_SOURCE)
  set(XOFFSET_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(XOFFSET_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(XOFFSET_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(XOFFSET_INSTALL OFF CACHE BOOL "" FORCE)
  set(XOFFSET_BUILD_VALIDATION_SCRATCH_RUNTIME ON CACHE BOOL "" FORCE)
  set(XOFFSET_BUILD_SNAPSHOT_INDEX_RUNTIME ON CACHE BOOL "" FORCE)
  add_subdirectory("${XC_SOURCE}" dependency)
else()
  find_package(XOffsetDatastructure CONFIG REQUIRED COMPONENTS full ValidationScratch snapshot_index_runtime)
endif()
enable_testing()
find_package(Threads REQUIRED)
foreach(peer IN ITEMS budget_peer_one budget_peer_two)
  add_library(${peer} SHARED ${peer}.cpp)
  target_link_libraries(${peer} PRIVATE XOffset::snapshot_index_runtime)
  set_target_properties(${peer} PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
endforeach()
foreach(probe IN ITEMS a b)
  add_library(scratch_probe_${probe} MODULE registry_probe_module.cpp)
  target_link_libraries(scratch_probe_${probe} PRIVATE XOffset::ValidationScratch)
  set_target_properties(scratch_probe_${probe} PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
endforeach()
add_executable(scratch_typed test_scratch_integration.cpp)
target_link_libraries(scratch_typed PRIVATE XOffset::xoffset XOffset::ValidationScratch)
add_executable(index_array test_snapshot_array_index.cpp)
target_link_libraries(index_array PRIVATE XOffset::xoffset XOffset::snapshot_index_runtime)
add_executable(index_budget test_snapshot_budget_shared.cpp)
target_link_libraries(index_budget PRIVATE XOffset::snapshot_index_runtime budget_peer_one budget_peer_two)
add_executable(scratch_contracts test_validation_scratch.cpp)
target_link_libraries(scratch_contracts PRIVATE XOffset::ValidationScratch Threads::Threads)
add_executable(scratch_two_tu test_registry_two_tu.cpp registry_two_tu_probe.cpp)
target_link_libraries(scratch_two_tu PRIVATE XOffset::ValidationScratch)
add_executable(scratch_modules test_registry_modules.cpp)
target_link_libraries(scratch_modules PRIVATE XOffset::ValidationScratch ${CMAKE_DL_LIBS})
add_dependencies(scratch_modules scratch_probe_a scratch_probe_b)
foreach(program IN ITEMS scratch_typed index_array index_budget scratch_contracts scratch_two_tu scratch_modules)
  target_compile_options(${program} PRIVATE -O3 -UNDEBUG)
  if(program STREQUAL "scratch_modules")
    add_test(NAME ${program} COMMAND ${program} "$<TARGET_FILE_NAME:scratch_probe_a>" "$<TARGET_FILE_NAME:scratch_probe_b>")
    set_tests_properties(${program} PROPERTIES WORKING_DIRECTORY "$<TARGET_FILE_DIR:scratch_modules>")
  else()
    add_test(NAME ${program} COMMAND ${program})
  endif()
  set_tests_properties(${program} PROPERTIES TIMEOUT 120)
endforeach()
