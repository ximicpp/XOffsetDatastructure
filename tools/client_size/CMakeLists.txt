# SC-14: what XOffset adds to a client app (tools/client_size/measure.sh drives it).
# Configure with a client toolchain (cmake/toolchains/*.cmake) and
# -DCMAKE_BUILD_TYPE=MinSizeRel; XOFFSET_CXX_STANDARD 20 or 23.
cmake_minimum_required(VERSION 3.20)
project(xoffset_client_size LANGUAGES CXX)

get_filename_component(XO_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
option(XO_SIZE_DEMO "Also build the demo client (demo/mapstate)" ON)
set(XOFFSET_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(XOFFSET_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(XOFFSET_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(XOFFSET_INSTALL OFF CACHE BOOL "" FORCE)
if(XO_SIZE_DEMO)
    set(XOFFSET_BUILD_DEMOS ON CACHE BOOL "" FORCE)
    set(MAPDEMO_CLIENT_ONLY ON CACHE BOOL "" FORCE)
endif()
add_subdirectory("${XO_REPO}" xoffset)

# Unused functions and data out of the program, as an app's release link does.
if(APPLE)
    set(XO_GC_LINK -Wl,-dead_strip)
else()
    set(XO_GC_LINK -Wl,--gc-sections)
endif()
foreach(n 0 1 2 4 8)
    foreach(variant exc noexc)
        set(t size_s${n}_${variant})
        add_executable(${t} size_client.cpp)
        target_compile_definitions(${t} PRIVATE XO_SCHEMAS=${n})
        target_compile_options(${t} PRIVATE -ffunction-sections -fdata-sections)
        target_link_options(${t} PRIVATE ${XO_GC_LINK})
        if(n GREATER 0)
            target_link_libraries(${t} PRIVATE XOffset::xoffset)
        else()
            target_compile_features(${t} PRIVATE cxx_std_${XOFFSET_CXX_STANDARD})
        endif()
        if(variant STREQUAL "noexc")
            xoffset_disable_exceptions(${t})
        endif()
    endforeach()
endforeach()
