cmake_minimum_required(VERSION 3.30)
project(ReceiveStandaloneConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(DEFINED XC_SOURCE)
  set(XOFFSET_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(XOFFSET_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(XOFFSET_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(XOFFSET_INSTALL OFF CACHE BOOL "" FORCE)
  add_subdirectory("${XC_SOURCE}" dependency)
else()
  find_package(XOffsetDatastructure CONFIG REQUIRED COMPONENTS full)
endif()
if(NOT XOffsetDatastructure_VERSION STREQUAL "0.1.0-alpha.1" AND NOT DEFINED XC_SOURCE)
  message(FATAL_ERROR "Exact complete alpha identity required")
endif()
enable_testing()
add_executable(consumer_receive_paths main.cpp raw_first.cpp core_first.cpp future_first.cpp)
target_link_libraries(consumer_receive_paths PRIVATE XOffset::xoffset)
target_compile_options(consumer_receive_paths PRIVATE -O3 -UNDEBUG)
add_test(NAME consumer_receive_paths COMMAND consumer_receive_paths)
set_tests_properties(consumer_receive_paths PROPERTIES TIMEOUT 120)
