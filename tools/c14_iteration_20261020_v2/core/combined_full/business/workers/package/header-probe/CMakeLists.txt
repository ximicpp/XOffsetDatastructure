cmake_minimum_required(VERSION 3.30)
project(ArchivePublicFirstIncludeObjects LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT DEFINED XC_SOURCE OR NOT DEFINED XC_OWN_HEADERS)
  message(FATAL_ERROR "Explicit clean source and own-header list required")
endif()
list(LENGTH XC_OWN_HEADERS header_count)
if(NOT header_count EQUAL 11)
  message(FATAL_ERROR "Exactly eleven own headers required")
endif()
set(XOFFSET_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(XOFFSET_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(XOFFSET_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(XOFFSET_INSTALL OFF CACHE BOOL "" FORCE)
set(XOFFSET_BUILD_VALIDATION_SCRATCH_RUNTIME OFF CACHE BOOL "" FORCE)
set(XOFFSET_BUILD_SNAPSHOT_INDEX_RUNTIME OFF CACHE BOOL "" FORCE)
add_subdirectory("${XC_SOURCE}" dependency EXCLUDE_FROM_ALL)
set(index 0)
foreach(header IN LISTS XC_OWN_HEADERS)
  add_library(public_header_${index} OBJECT "probe_${index}.cpp")
  target_link_libraries(public_header_${index} PRIVATE XOffset::xoffset)
  target_compile_options(public_header_${index} PRIVATE -O3 -UNDEBUG)
  math(EXPR index "${index}+1")
endforeach()
